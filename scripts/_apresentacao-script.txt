<script>
const slides=["DATA", "DATA", "DATA", "DATA", "DATA", "DATA"];let idx=0,ctx=null,muted=false,playing=false,notesTimer=null;const img=document.getElementById('slide');function render(){img.src=slides[idx];img.alt='Página '+(idx+1)+' da apresentação AMPLIGUARD';document.getElementById('count').textContent=(idx+1)+' / '+slides.length;document.getElementById('next').textContent=idx===slides.length-1?'Recomeçar ↻':'Avançar ›';}function startMusic(){if(muted)return;if(!ctx){ctx=new(window.AudioContext||window.webkitAudioContext)();}ctx.resume();if(playing)return;playing=true;let n=0;const chords=[[196,246.94,293.66],[174.61,220,261.63],[164.81,196,246.94],[174.61,220,293.66]];function play(){if(!ctx||muted||ctx.state==='closed')return;let t=ctx.currentTime+.02;let c=chords[n++%chords.length];c.forEach((f,j)=>{let o=ctx.createOscillator(),g=ctx.createGain();o.type='sine';o.frequency.value=f;g.gain.setValueAtTime(0,t);g.gain.linearRampToValueAtTime(.018,t+.35);g.gain.setValueAtTime(.018,t+1.4);g.gain.exponentialRampToValueAtTime(.0001,t+2.9);o.connect(g);g.connect(ctx.destination);o.start(t);o.stop(t+3);});}play();notesTimer=setInterval(play,2600)}function go(d){startMusic();idx=(idx+d+slides.length)%slides.length;render();}img.addEventListener('click',()=>go(1));img.addEventListener('keydown',e=>{if(e.key==='Enter'||e.key===' '){e.preventDefault();go(1)}});document.getElementById('next').onclick=()=>go(1);document.getElementById('prev').onclick=()=>go(-1);document.getElementById('music').onclick=()=>{muted=!muted;document.getElementById('music').textContent=muted?'♫ Música: desligada':'♫ Música: ligada';if(muted){clearInterval(notesTimer);playing=false;if(ctx)ctx.suspend()}else startMusic()};document.addEventListener('keydown',e=>{if(e.key==='ArrowRight')go(1);if(e.key==='ArrowLeft')go(-1)});render();</script></body></html>
